==================================================
CYBERSENTINEL ANALYTICS — DAX MEASURES
==================================================


--> GLOBAL / SECURITY EVENTS
==================================================

Total Events =
COUNTROWS('warehouse fact_security_event')

Critical Events =
CALCULATE(
    [Total Events],
    'warehouse fact_security_event'[severity] = "Critical"
)

High Events =
CALCULATE(
    [Total Events],
    'warehouse fact_security_event'[severity] = "High"
)

Medium Events =
CALCULATE(
    [Total Events],
    'warehouse fact_security_event'[severity] = "Medium"
)

Low Events =
CALCULATE(
    [Total Events],
    'warehouse fact_security_event'[severity] = "Low"
)

Machines Affectées =
DISTINCTCOUNT(
    'warehouse fact_security_event'[machine_key]
)

Sources de données =
DISTINCTCOUNT(
    'warehouse fact_security_event'[event_source]
)

Utilisateurs actifs =
DISTINCTCOUNT(
    'warehouse fact_security_event'[user_key]
)


==================================================
--> AUTHENTIFICATION
==================================================

Tentatives totales =
COUNTROWS('warehouse fact_authentication_event')

Connexions réussies =
CALCULATE(
    COUNTROWS('warehouse fact_authentication_event'),
    'warehouse fact_authentication_event'[is_success] = TRUE()
)

Connexions échouées =
CALCULATE(
    COUNTROWS('warehouse fact_authentication_event'),
    'warehouse fact_authentication_event'[is_failure] = TRUE()
)

Connexions à distance =
CALCULATE(
    COUNTROWS('warehouse fact_authentication_event'),
    'warehouse fact_authentication_event'[is_remote_logon] = TRUE()
)

Connexions privilégiées =
CALCULATE(
    COUNTROWS('warehouse fact_authentication_event'),
    'warehouse fact_authentication_event'[is_privileged_logon] = TRUE()
)

Utilisateurs uniques =
DISTINCTCOUNT(
    'warehouse fact_authentication_event'[user_key]
)

Authentication Success Rate =
DIVIDE(
    [Connexions réussies],
    [Tentatives totales],
    0
)

Failed Login Rate =
DIVIDE(
    [Connexions échouées],
    [Tentatives totales],
    0
)


==================================================
--> ENDPOINTS / MACHINES
==================================================

Total Machines =
DISTINCTCOUNT(
    'warehouse dim_machine'[machine_id]
)

Machines Actives =
CALCULATE(
    DISTINCTCOUNT('warehouse dim_machine'[machine_id]),
    'warehouse dim_machine'[machine_status] = "Active"
)

Machines Inactives =
CALCULATE(
    DISTINCTCOUNT('warehouse dim_machine'[machine_id]),
    'warehouse dim_machine'[machine_status] = "Inactive"
)

OS différents =
COALESCE(
    CALCULATE(
        DISTINCTCOUNT(
            'warehouse dim_machine'[windows_version]
        ),
        NOT ISBLANK(
            'warehouse dim_machine'[windows_version]
        )
    ),
    0
)

Santé Moyenne =
AVERAGE(
    'warehouse fact_machine_health_snapshot'[health_score]
)

Machines Critiques =
CALCULATE(
    DISTINCTCOUNT(
        'warehouse fact_security_event'[machine_key]
    ),
    'warehouse fact_security_event'[severity] = "Critical"
)


==================================================
--> MENACES / SOC
==================================================

Total Security Events =[Total Events]

Authentication Failure =

    [Connexions échouées],

    'warehouse fact_defender_event'

        COUNTROWS(
            'warehouse fact_usb_event'
        ),
        'warehouse fact_usb_event'[is_connected] = TRUE()
    ),
    0
)

--> NETWORK

Connexions totales =

)

==================================================
--> DATA QUALITY & PIPELINE
==================================================

Fichiers collectés =
COUNTROWS('warehouse fact_file_quality')

Fichiers validés =
CALCULATE(
    COUNTROWS(
        'warehouse fact_file_quality'
    ),
    'warehouse fact_file_quality'[validation_status] = "PASS"
)

Échecs de validation =
COALESCE(
    CALCULATE(
        COUNTROWS(
            'warehouse fact_file_quality'
        ),
        'warehouse fact_file_quality'[validation_status] = "FAIL"
    ),
    0
)

Taux de qualité =
DIVIDE(
    [Fichiers validés],
    [Fichiers collectés],
    0
)

Sources opérationnelles =
CALCULATE(
    DISTINCTCOUNT(
        'warehouse fact_collection_source_status'[source_name]
    ),
    'warehouse fact_collection_source_status'[source_status] = "SUCCESS"
)

Dernière exécution pipeline =
MAX(
    'warehouse fact_collection_source_status'[collection_timestamp]
)
